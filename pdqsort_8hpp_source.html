112 // Attempts to use insertion sort on [begin, end). Will return false if more than
113 // partial_insertion_sort_limit elements were moved, and abort sorting. Otherwise it will
114 // successfully sort and return true.
115 template<class Iter, class Compare>
116 inline bool partial_insertion_sort(Iter begin, Iter end, Compare comp) {
117 typedef typename boost::movelib::iterator_traits<Iter>::value_type T;
118 typedef typename boost::movelib:: iter_size<Iter>::type size_type;
119 if (begin == end) return true;
120
121 size_type limit = 0;
122 for (Iter cur = begin + 1; cur != end; ++cur) {
123 if (limit > partial_insertion_sort_limit) return false;
124
125 Iter sift = cur;
126 Iter sift_1 = cur - 1;
127
128 // Compare first so we can avoid 2 moves for an element already positioned correctly.
129 if (comp(*sift, *sift_1)) {
130 T tmp = boost::move(*sift);
131
132 do { *sift-- = boost::move(*sift_1); }
133 while (sift != begin && comp(tmp, *--sift_1));
134
135 *sift = boost::move(tmp);
136 limit += size_type(cur - sift);
137 }
138 }
139
140 return true;
141 }

143 template<class Iter, class Compare>
144 inline void sort2(Iter a, Iter b, Compare comp) {
145 if (comp(*b, *a)) boost::adl_move_iter_swap(a, b);
146 }

148 // Sorts the elements *a, *b and *c using comparison function comp.
149 template<class Iter, class Compare>
150 inline void sort3(Iter a, Iter b, Iter c, Compare comp) {
151 sort2(a, b, comp);
152 sort2(b, c, comp);
153 sort2(a, b, comp);

156 // Partitions [begin, end) around pivot *begin using comparison function comp. Elements equal
157 // to the pivot are put in the right-hand partition. Returns the position of the pivot after
158 // partitioning and whether the passed sequence already was correctly partitioned. Assumes the
159 // pivot is a median of at least 3 elements and that [begin, end) is at least
160 // insertion_sort_threshold long.
161 template<class Iter, class Compare>
162 pdqsort_detail::pair<Iter, bool> partition_right(Iter begin, Iter end, Compare comp) {
163 typedef typename boost::movelib::iterator_traits<Iter>::value_type T;
164
165 // Move pivot into local for speed.
166 T pivot(boost::move(*begin));
167
168 Iter first = begin;
169 Iter last = end;
170
171 // Find the first element greater than or equal than the pivot (the median of 3 guarantees
172 // this exists).
173 while (comp(*++first, pivot));
174
175 // Find the first element strictly smaller than the pivot. We have to guard this search if
176 // there was no element before *first.
177 if (first - 1 == begin) while (first < last && !comp(*--last, pivot));
178 else while ( !comp(*--last, pivot));
179
180 // If the first pair of elements that should be swapped to partition are the same element,
181 // the passed in sequence already was correctly partitioned.
182 bool already_partitioned = first >= last;
183
184 // Keep swapping pairs of elements that are on the wrong side of the pivot. Previously
185 // swapped pairs guard the searches, which is why the first iteration is special-cased
186 // above.
187 while (first < last) {
188 boost::adl_move_iter_swap(first, last);
189 while (comp(*++first, pivot));
190 while (!comp(*--last, pivot));
191 }
192
193 // Put the pivot in the right place.
194 Iter pivot_pos = first - 1;
195 if(begin != pivot_pos) //Avoid potential self-move
196 *begin = boost::move(*pivot_pos);
197 *pivot_pos = boost::move(pivot);
198
199 return pdqsort_detail::pair<Iter, bool>(pivot_pos, already_partitioned);
200 }
201
202 // Similar function to the one above, except elements equal to the pivot are put to the left of
203 // the pivot and it doesn't check or return if the passed sequence already was partitioned.
204 // Since this is rarely used (the many equal case), and in that case pdqsort already has O(n)
205 // performance, no block quicksort is applied here for simplicity.
206 template<class Iter, class Compare>
207 inline Iter partition_left(Iter begin, Iter end, Compare comp) {
208 typedef typename boost::movelib::iterator_traits<Iter>::value_type T;
209
210 T pivot(boost::move(*begin));
211 Iter first = begin;
212 Iter last = end;
213
214 while (comp(pivot, *--last));
215
216 if (last + 1 == end) while (first < last && !comp(pivot, *++first));
217 else while ( !comp(pivot, *++first));
218
219 while (first < last) {
220 boost::adl_move_iter_swap(first, last);
221 while (comp(pivot, *--last));
222 while (!comp(pivot, *++first));
223 }
224
225 Iter pivot_pos = last;
226 *begin = boost::move(*pivot_pos);
227 *pivot_pos = boost::move(pivot);
228
229 return pivot_pos;
230 }

233 template<class Iter, class Compare>
234 void pdqsort_loop( Iter begin, Iter end, Compare comp
235 , typename boost::movelib:: iter_size<Iter>::type bad_allowed
236 , bool leftmost = true)
237 {
238 typedef typename boost::movelib:: iter_size<Iter>::type size_type;
239
240 // Use a while loop for tail recursion elimination.
241 while (true) {
242 size_type size = size_type(end - begin);
243
244 // Insertion sort is faster for small arrays.
245 if (size < insertion_sort_threshold) {
246 insertion_sort(begin, end, comp);
247 return;
248 }
249
250 // Choose pivot as median of 3 or pseudomedian of 9.
251 size_type s2 = size / 2;
252 if (size > ninther_threshold) {
253 sort3(begin, begin + s2, end - 1, comp);
254 sort3(begin + 1, begin + (s2 - 1), end - 2, comp);
255 sort3(begin + 2, begin + (s2 + 1), end - 3, comp);
256 sort3(begin + (s2 - 1), begin + s2, begin + (s2 + 1), comp);
257 boost::adl_move_iter_swap(begin, begin + s2);
258 } else sort3(begin + s2, begin, end - 1, comp);
259
260 // If *(begin - 1) is the end of the right partition of a previous partition operation
261 // there is no element in [begin, end) that is smaller than *(begin - 1). Then if our
262 // pivot compares equal to *(begin - 1) we change strategy, putting equal elements in
263 // the left partition, greater elements in the right partition. We do not have to
264 // recurse on the left partition, since it's sorted (all equal).
265 if (!leftmost && !comp(*(begin - 1), *begin)) {
266 begin = partition_left(begin, end, comp) + 1;
267 continue;
268 }
269
270 // Partition and get results.
271 pdqsort_detail::pair<Iter, bool> part_result = partition_right(begin, end, comp);
272 Iter pivot_pos = part_result.first;
273 bool already_partitioned = part_result.second;
274
275 // Check for a highly unbalanced partition.
276 size_type l_size = size_type(pivot_pos - begin);
277 size_type r_size = size_type(end - (pivot_pos + 1));
278 bool highly_unbalanced = l_size < size / 8 || r_size < size / 8;
279
280 // If we got a highly unbalanced partition we shuffle elements to break many patterns.
281 if (highly_unbalanced) {
282 // If we had too many bad partitions, switch to heapsort to guarantee O(n log n).
283 if (--bad_allowed == 0) {
284 boost::movelib::heap_sort(begin, end, comp);
285 return;
286 }
287
288 if (l_size >= insertion_sort_threshold) {
289 boost::adl_move_iter_swap(begin, begin + l_size / 4);
290 boost::adl_move_iter_swap(pivot_pos - 1, pivot_pos - l_size / 4);
291
292 if (l_size > ninther_threshold) {
293 boost::adl_move_iter_swap(begin + 1, begin + (l_size / 4 + 1));
294 boost::adl_move_iter_swap(begin + 2, begin + (l_size / 4 + 2));
295 boost::adl_move_iter_swap(pivot_pos - 2, pivot_pos - (l_size / 4 + 1));
296 boost::adl_move_iter_swap(pivot_pos - 3, pivot_pos - (l_size / 4 + 2));
297 }
298 }
299
300 if (r_size >= insertion_sort_threshold) {
301 boost::adl_move_iter_swap(pivot_pos + 1, pivot_pos + (1 + r_size / 4));
302 boost::adl_move_iter_swap(end - 1, end - r_size / 4);
303
304 if (r_size > ninther_threshold) {
305 boost::adl_move_iter_swap(pivot_pos + 2, pivot_pos + (2 + r_size / 4));
306 boost::adl_move_iter_swap(pivot_pos + 3, pivot_pos + (3 + r_size / 4));
307 boost::adl_move_iter_swap(end - 2, end - (1 + r_size / 4));
308 boost::adl_move_iter_swap(end - 3, end - (2 + r_size / 4));
309 }
310 }
311 } else {
312 // If we were decently balanced and we tried to sort an already partitioned
313 // sequence try to use insertion sort.
314 if (already_partitioned && partial_insertion_sort(begin, pivot_pos, comp)
315 && partial_insertion_sort(pivot_pos + 1, end, comp)) return;
316 }
317
318 // Sort the left partition first using recursion and do tail recursion elimination for
319 // the right-hand partition.
320 pdqsort_loop<Iter, Compare>(begin, pivot_pos, comp, bad_allowed, leftmost);
321 begin = pivot_pos + 1;
322 leftmost = false;
323 }
324 }
325}

329void pdqsort(Iter begin, Iter end, Compare comp)
330{
331 if (begin == end) return;
332 typedef typename boost::movelib:: iter_size<Iter>::type size_type;
333 pdqsort_detail::pdqsort_loop<Iter, Compare>(begin, end, comp, pdqsort_detail::log2(size_type(end - begin)));
334}
