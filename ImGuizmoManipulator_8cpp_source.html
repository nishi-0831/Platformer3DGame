204void mtgb::ImGuizmoManipulator::UpdateManpulator()
205{
206 isUsing_ = ImGuizmo::IsUsing();
207
208 // ギズモを使用 (動かしていなくても長押しを使用状態とみなす)
209 if (wasUsing_ == false && isUsing_ == true)
210 {
211 if (pTargetTransform_ == nullptr)
212 return;
213 pTargetPrevTransformMemento_ = pTargetTransform_->SaveToMemento();
214 }
215
216 // ギズモの使用を終了
217 if (wasUsing_ == true && isUsing_ == false)
218 {
219 if (pTargetPrevTransformMemento_ == nullptr || pTargetTransform_ == nullptr)
220 return;
221
222 TransformMemento* memento = pTargetTransform_->SaveToMemento();
223 GuizmoManipulateCommand* cmd = new GuizmoManipulateCommand(pTargetPrevTransformMemento_, memento);
224 Game::System<CommandHistoryManager>().ExecuteCommand(cmd);
225
226 // メモリの解放はコマンドが行うと断定して、ポインタを空にする
227 pTargetPrevTransformMemento_ = nullptr;
228 }
229
230 wasUsing_ = isUsing_;
231}
