47 // It may seem odd to use such small pages.
48 //
49 // However, on a typical Windows implementation that uses
50 // the OS allocator, "normal size" pages interact with the
51 // "ordinary" operator new, slowing it down dramatically.
52 //
53 // 512 byte pages are handled by the small object allocator,
54 // and don't interfere with ::new.
55 //
56 // The other alternative is to use much bigger pages (1M.)
57 //
58 // It is surprisingly easy to hit pathological behavior by
59 // varying the page size. g++ 2.96 on Red Hat Linux 7.2,
60 // for example, passionately dislikes 496. 512 seems OK.
61

70#endif

89 static inline void * alloc()
90 {
91#ifdef BOOST_HAS_THREADS
92 lightweight_mutex::scoped_lock lock( mutex() );
93#endif
94 if(block * x = free)
95 {
96 free = x->next;
97 return x;
98 }
99 else
100 {
101 if(last == items_per_page)
102 {
103 // "Listen to me carefully: there is no memory leak"
104 // -- Scott Meyers, Eff C++ 2nd Ed Item 10
105 page = ::new block[items_per_page];
106 last = 0;
107 }
108
109 return &page[last++];
110 }
111 }
