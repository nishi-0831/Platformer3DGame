45// Tests whether type Iterator satisfies the requirements for a
46// TrivialIterator.
47// Preconditions: i != j, *i == val
48template <class Iterator, class T>
49void trivial_iterator_test(const Iterator i, const Iterator j, T val)
50{
51 Iterator k;
52 BOOST_TEST(i == i);
53 BOOST_TEST(j == j);
54 BOOST_TEST(i != j);
55#ifdef BOOST_NO_STD_ITERATOR_TRAITS
56 T v = *i;
57#else
58 typename std::iterator_traits<Iterator>::value_type v = *i;
59#endif
60 BOOST_TEST(v == val);
61 boost::ignore_unused(v);
62#if 0
63 // hmm, this will give a warning for transform_iterator... perhaps
64 // this should be separated out into a stand-alone test since there
65 // are several situations where it can't be used, like for
66 // integer_range::iterator.
67 BOOST_TEST(v == i->foo());
68#endif
69 k = i;
70 BOOST_TEST(k == k);
71 BOOST_TEST(k == i);
72 BOOST_TEST(k != j);
73 BOOST_TEST(*k == val);
74 boost::ignore_unused(k);
75}

87// Preconditions: *i == v1, *++i == v2
88template <class Iterator, class T>
89void input_iterator_test(Iterator i, T v1, T v2)
90{
91 Iterator i1(i);
92
93 BOOST_TEST(i == i1);
94 BOOST_TEST(!(i != i1));
95
96 // I can see no generic way to create an input iterator
97 // that is in the domain of== of i and != i.
98 // The following works for istream_iterator but is not
99 // guaranteed to work for arbitrary input iterators.
100 //
101 // Iterator i2;
102 //
103 // BOOST_TEST(i != i2);
104 // BOOST_TEST(!(i == i2));
105
106 BOOST_TEST(*i1 == v1);
107 BOOST_TEST(*i == v1);
108
109 // we cannot test for equivalence of (void)++i & (void)i++
110 // as i is only guaranteed to be single pass.
111 BOOST_TEST(*i++ == v1);
112 boost::ignore_unused(i1);
113
114 i1 = i;
115
116 BOOST_TEST(i == i1);
117 BOOST_TEST(!(i != i1));
118
119 BOOST_TEST(*i1 == v2);
120 BOOST_TEST(*i == v2);
121 boost::ignore_unused(i1);
122
123 // i is dereferencable, so it must be incrementable.
124 ++i;
125
126 // how to test for operator-> ?
127}

132template <bool is_pointer> struct lvalue_test
133{
134 template <class Iterator> static void check(Iterator)
135 {
136# ifndef BOOST_NO_STD_ITERATOR_TRAITS
137 typedef typename std::iterator_traits<Iterator>::reference reference;
138 typedef typename std::iterator_traits<Iterator>::value_type value_type;
139# else
140 typedef typename Iterator::reference reference;
141 typedef typename Iterator::value_type value_type;
142# endif
143 static_assert(std::is_reference<reference>::value, "reference must be a reference type.");
144 static_assert(
145 std::is_same<reference, value_type&>::value || std::is_same<reference, const value_type&>::value,
146 "reference must either be a reference to value_type or constant reference to value_type."
147 );
148 }

157template <class Iterator, class T>
158void forward_iterator_test(Iterator i, T v1, T v2)
159{
160 input_iterator_test(i, v1, v2);
161
162 Iterator i1 = i, i2 = i;
163
164 BOOST_TEST(i == i1++);
165 BOOST_TEST(i != ++i2);
166
167 trivial_iterator_test(i, i1, v1);
168 trivial_iterator_test(i, i2, v1);
169
170 ++i;
171 BOOST_TEST(i == i1);
172 BOOST_TEST(i == i2);
173 ++i1;
174 ++i2;
175
176 trivial_iterator_test(i, i1, v2);
177 trivial_iterator_test(i, i2, v2);
178
179 // borland doesn't allow non-type template parameters
180# if !defined(BOOST_BORLANDC) || (BOOST_BORLANDC > 0x551)
181 lvalue_test<std::is_pointer<Iterator>::value>::check(i);
182#endif
183}

185// Preconditions: *i == v1, *++i == v2
186template <class Iterator, class T>
187void bidirectional_iterator_test(Iterator i, T v1, T v2)
188{
189 forward_iterator_test(i, v1, v2);
190 ++i;
191
192 Iterator i1 = i, i2 = i;
193
194 BOOST_TEST(i == i1--);
195 BOOST_TEST(i != --i2);
196
197 trivial_iterator_test(i, i1, v2);
198 trivial_iterator_test(i, i2, v2);
199
200 --i;
201 BOOST_TEST(i == i1);
202 BOOST_TEST(i == i2);
203 ++i1;
204 ++i2;
205
206 trivial_iterator_test(i, i1, v1);
207 trivial_iterator_test(i, i2, v1);
208}

214// Preconditions: [i,i+N) is a valid range
215template <class Iterator, class TrueVals>
216void random_access_iterator_test(Iterator i, int N, TrueVals vals)
217{
218 bidirectional_iterator_test(i, vals[0], vals[1]);
219 const Iterator j = i;
220 int c;
221
222 typedef typename std::iterator_traits<Iterator>::value_type value_type;
223 struct local
224 {
225 static value_type to_value_type(value_type v) { return v; }
226 };
227
228 for (c = 0; c < N-1; ++c) {
229 BOOST_TEST(i == j + c);
230 BOOST_TEST(*i == vals[c]);
231 BOOST_TEST(*i == local::to_value_type(j[c]));
232 BOOST_TEST(*i == *(j + c));
233 BOOST_TEST(*i == *(c + j));
234 ++i;
235 BOOST_TEST(i > j);
236 BOOST_TEST(i >= j);
237 BOOST_TEST(j <= i);
238 BOOST_TEST(j < i);
239 }
240
241 Iterator k = j + N - 1;
242 for (c = 0; c < N-1; ++c) {
243 BOOST_TEST(i == k - c);
244 BOOST_TEST(*i == vals[N - 1 - c]);
245 BOOST_TEST(*i == local::to_value_type(j[N - 1 - c]));
246 Iterator q = k - c;
247 boost::ignore_unused(q);
248 BOOST_TEST(*i == *q);
249 BOOST_TEST(i > j);
250 BOOST_TEST(i >= j);
251 BOOST_TEST(j <= i);
252 BOOST_TEST(j < i);
253 --i;
254 }
255}
