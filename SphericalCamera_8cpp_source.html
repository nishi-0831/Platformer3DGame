29 mtgb::Mathf::SphericalCoord CameraCartesianToSpherical(const Vector3& _offset)
30 {
31 float x = _offset.x;
32 float y = _offset.y;
33 float z = _offset.z;
34
35 float r = sqrtf(x * x + y * y + z * z);
36
37 // カメラの座標系に合わせた変換
38 // Y軸が上を向いており、-Y方向にcosθが向いているため
39 float theta = std::acosf(-y / r); // カメラ用の極角

41
42 return Mathf::SphericalCoord { .r = r, .theta = theta, .phi = phi };

182{
183 Vector3 movement;

212 azimuthalAngleRad_ = NormalizeAngleRad(azimuthalAngleRad_);
213
214 // カメラを動かす
215 MoveCameraSpherical(distance_);

243void mtgb::SphericalCamera::MoveCameraSpherical(float _distance)
244{
245 if (pTargetTransform_ == nullptr)
246 return;
247
248 // 注視点を計算
249 float lerpedY = std::lerp(baseY_, distY_, lookAtPosLerpProgress_);
250 Vector3 lookAtTarget =
251 Vector3(pTargetTransform_->position.x, lerpedY, pTargetTransform_->position.z) + lookAtPositionOffset_;
252
253 // 球面座標で新しいカメラ位置を計算
254 float theta = polarAngleRad_;
255 float phi = azimuthalAngleRad_;
256
257 Vector3 offset;
258 offset.x = _distance * sinf(theta) * cosf(phi);
259 offset.y = -_distance * cosf(theta);
260 offset.z = -_distance * sinf(theta) * sinf(phi);
261
262 Vector3 newCameraPos = lookAtTarget + offset;
263
264 pCameraTransform_->position = newCameraPos;
265
266 // カメラの回転を設定
267 Vector3 lookDir = lookAtTarget - pCameraTransform_->position;
268 pCameraTransform_->rotate = Quaternion::LookRotation(lookDir.Normalize(), Vector3::Up());
269}
