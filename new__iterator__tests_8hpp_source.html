118template <class Iterator, class T>
119void constant_lvalue_iterator_test(Iterator i, T v1)
120{
121 Iterator i2(i);
122 typedef typename std::iterator_traits<Iterator>::value_type value_type;
123 typedef typename std::iterator_traits<Iterator>::reference reference;
124 static_assert(std::is_same<const value_type&, reference>::value,
125 "reference type must be the same as const value_type& for "
126 "constant lvalue iterator.");
127 const T& v2 = *i2;
128 BOOST_TEST(v1 == v2);
129#ifndef BOOST_NO_LVALUE_RETURN_DETECTION
130 static_assert(is_lvalue_iterator<Iterator>::value
131 && !is_non_const_lvalue_iterator<Iterator>::value,
132 "Iterator must be a const lvalue iterator.");
133#endif
134}

137void non_const_lvalue_iterator_test(Iterator i, T v1, T v2)
138{
139 Iterator i2(i);
140 typedef typename std::iterator_traits<Iterator>::value_type value_type;
141 typedef typename std::iterator_traits<Iterator>::reference reference;
142 static_assert(std::is_same<value_type&, reference>::value,
143 "reference type must be the same as value_type& for "
144 "non-constant lvalue iterator.");

191// bidirectional
192// Preconditions: *i == v1, *++i == v2
193template <class Iterator, class T>
194void bidirectional_readable_iterator_test(Iterator i, T v1, T v2)
195{
196 Iterator j(i);
197 ++j;
198 forward_readable_iterator_test(i, j, v1, v2);
199 ++i;
200
201 Iterator i1 = i, i2 = i;
202
203 BOOST_TEST(i == i1--);
204 BOOST_TEST(i != --i2);
205
206 readable_iterator_test(i, v2);
207 readable_iterator_test(i1, v1);
208 readable_iterator_test(i2, v1);
209
210 --i;
211 BOOST_TEST(i == i1);
212 BOOST_TEST(i == i2);
213 ++i1;
214 ++i2;
215
216 readable_iterator_test(i, v1);
217 readable_iterator_test(i1, v2);
218 readable_iterator_test(i2, v2);

221// random access
222// Preconditions: [i,i+N) is a valid range
223template <class Iterator, class TrueVals>
224void random_access_readable_iterator_test(Iterator i, int N, TrueVals vals)
225{
226 bidirectional_readable_iterator_test(i, vals[0], vals[1]);
227 const Iterator j = i;
228 int c;
229
230 for (c = 0; c < N - 1; ++c)
231 {
232 BOOST_TEST(i == j + c);
233 BOOST_TEST(*i == vals[c]);
234 typename std::iterator_traits<Iterator>::value_type x = j[c];
235 BOOST_TEST(*i == x);
236 BOOST_TEST(*i == *(j + c));
237 BOOST_TEST(*i == *(c + j));
238 ++i;
239 BOOST_TEST(i > j);
240 BOOST_TEST(i >= j);
241 BOOST_TEST(j <= i);
242 BOOST_TEST(j < i);
243 }
244
245 Iterator k = j + N - 1;
246 for (c = 0; c < N - 1; ++c)
247 {
248 BOOST_TEST(i == k - c);
249 BOOST_TEST(*i == vals[N - 1 - c]);
250 typename std::iterator_traits<Iterator>::value_type x = j[N - 1 - c];
251 BOOST_TEST(*i == x);
252 Iterator q = k - c;
253 BOOST_TEST(*i == *q);
254 BOOST_TEST(i > j);
255 BOOST_TEST(i >= j);
256 BOOST_TEST(j <= i);
257 BOOST_TEST(j < i);
258 --i;
259 }
260}
