18//
19// Traversal Categories
20//
21struct no_traversal_tag {};
22struct incrementable_traversal_tag : public no_traversal_tag {};
23struct single_pass_traversal_tag : public incrementable_traversal_tag {};
24struct forward_traversal_tag : public single_pass_traversal_tag {};
25struct bidirectional_traversal_tag : public forward_traversal_tag {};
26struct random_access_traversal_tag : public bidirectional_traversal_tag {};
27
28//
29// Convert an iterator category into a traversal tag
30//
31template< typename Cat >
32using iterator_category_to_traversal_t = mp11::mp_cond<
33 // if already convertible to a traversal tag, we're done.
34 std::is_convertible< Cat, incrementable_traversal_tag >, Cat,
35 std::is_convertible< Cat, std::random_access_iterator_tag >, random_access_traversal_tag,
36 std::is_convertible< Cat, std::bidirectional_iterator_tag >, bidirectional_traversal_tag,
37 std::is_convertible< Cat, std::forward_iterator_tag >, forward_traversal_tag,
38 std::is_convertible< Cat, std::input_iterator_tag >, single_pass_traversal_tag,
39 std::is_convertible< Cat, std::output_iterator_tag >, incrementable_traversal_tag,
40 std::true_type, void
41>;
42
43template< typename Cat >
44struct iterator_category_to_traversal
45{
46 using type = iterator_category_to_traversal_t< Cat >;
47};
48
49// Trait to get an iterator's traversal category
50template< typename Iterator >
51using iterator_traversal_t = iterator_category_to_traversal_t<
52 typename std::iterator_traits< Iterator >::iterator_category
53>;
54
55template< typename Iterator = mpl::arg< 1 > >
56struct iterator_traversal
57{
58 using type = iterator_traversal_t< Iterator >;
59};
60
61//
62// Convert an iterator traversal to one of the traversal tags.
63//
64template< typename Traversal >
65using pure_traversal_tag_t = mp11::mp_cond<
66 std::is_convertible< Traversal, random_access_traversal_tag >, random_access_traversal_tag,
67 std::is_convertible< Traversal, bidirectional_traversal_tag >, bidirectional_traversal_tag,
68 std::is_convertible< Traversal, forward_traversal_tag >, forward_traversal_tag,
69 std::is_convertible< Traversal, single_pass_traversal_tag >, single_pass_traversal_tag,
70 std::is_convertible< Traversal, incrementable_traversal_tag >, incrementable_traversal_tag,
71 std::true_type, void
72>;
73
74template< typename Traversal >
75struct pure_traversal_tag
76{
77 using type = pure_traversal_tag_t< Traversal >;
78};
79
80//
81// Trait to retrieve one of the iterator traversal tags from the iterator category or traversal.
82//
83template< typename Iterator >
84using pure_iterator_traversal_t = pure_traversal_tag_t<
85 iterator_traversal_t< Iterator >
86>;
87
88template< typename Iterator = mpl::arg< 1 > >
89struct pure_iterator_traversal
90{
91 using type = pure_iterator_traversal_t< Iterator >;
92};
