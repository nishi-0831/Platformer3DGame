180void mtgb::Audio::UpdateEmitter()
181{
182 if (pListenerTransform_ == nullptr)
183 return;
184 XAUDIO2_VOICE_DETAILS voiceDetails;
185 pMasteringVoice_->GetVoiceDetails(&voiceDetails);
186
187 for (auto& [name, emitter] : emitterMap_)
188 {
189 AudioClip* clip = audioClipMap_[name];
190
191 X3DAUDIO_DSP_SETTINGS dspSettings;
192 // 出力先となるマスターボイスのチャネル数
193 dspSettings.DstChannelCount = voiceDetails.InputChannels;
194 // ソースとなるクリップのチャネル数
195 dspSettings.SrcChannelCount = clip->pWaveData_->waveFormat.nChannels;
196
197 std::vector<FLOAT32> mat(dspSettings.SrcChannelCount * dspSettings.DstChannelCount);
198 dspSettings.pMatrixCoefficients = mat.data();
199
200 Transform& emitterTransform = Game::System<TransformCP>().Get(emitter.emitterId);
201
202 X3DAUDIO_EMITTER& pX3DEmitter = emitter.x3DEmitter;
203 pX3DEmitter.Position = emitterTransform.GetWorldPosition();
204 pX3DEmitter.OrientFront = emitterTransform.Forward();
205 pX3DEmitter.OrientTop = emitterTransform.Up();
206 // エミッターの座標を原点として、各チャネルが広がる半径
207 pX3DEmitter.ChannelRadius = 1.0f;
208 // エミッターの前方ベクトルを基準とした角度
209 // 左右に分けるために、-π/2とπ/2を設定
210 std::vector<FLOAT32> channelAzimuths = { -X3DAUDIO_2PI / 4.0f, X3DAUDIO_2PI / 4.0f };
211 pX3DEmitter.pChannelAzimuths = channelAzimuths.data();
212
213 X3DAudioCalculate(x3DInstance_, &listener_, &emitter.x3DEmitter, X3DAUDIO_CALCULATE_MATRIX, &dspSettings);
214 clip->pSourceVoice_->SetOutputMatrix(
215 pMasteringVoice_,
216 dspSettings.SrcChannelCount,
217 dspSettings.DstChannelCount,
218 dspSettings.pMatrixCoefficients
219 );
220 }
221}
