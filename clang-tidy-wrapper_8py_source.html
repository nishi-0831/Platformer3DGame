14def extract_compile_commands_dir(args):
15 """ extract the compile_commands.json directory specified with -p from the arguments"""
16 for i, arg in enumerate(args):
17 if arg.startswith('-p='):
18 return arg[3:]
19 elif arg == '-p' and i + 1 < len(args):
20 return args[i + 1]
21 return None

34def generate_compile_commands_with_header_files(compile_commands_path, compile_commands_json):
35 # pythonのデフォルトのエンコーディングを変更する
36 env = os.environ.copy()

39 cmds = [
40 ['compdb','-p',compile_commands_path,'list']
41 ]
42
43 for cmd in cmds:
44 try:
45 proc = subprocess.run(cmd, env=env, check=True,stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
46 out = proc.stdout
47 if out:
48 os.makedirs(os.path.dirname(compile_commands_json),exist_ok=True)
49 with open(compile_commands_json,'w') as f:
50 f.write(out)
51 print(f"[clang-tidy-wrapper] compdb produced compile_commands.json ({len(out)} bytes)", file=sys.stderr)
52 return True
53 except FileNotFoundError:
54 # compdbコマンドが見つからない
55 continue
56 except subprocess.CalledProcessError as e:
57 continue
