28void mtgb::Transform::Compute()
29{
30 using DirectX::XMConvertToRadians;
31 using DirectX::XMMatrixRotationQuaternion;
32 using DirectX::XMMatrixScaling;
33 using DirectX::XMMatrixTranslation;
34 using DirectX::XMQuaternionNormalize;
35
36 // 親の行列を取得
37 Matrix4x4 parentMat {};
38 GenerateParentMatrix(&parentMat);
39
40 // 現在のワールド座標から行列を作成
41 Matrix4x4 currWorldMat = XMMatrixScaling(scale.x, scale.y, scale.z) *
42 XMMatrixRotationQuaternion(XMQuaternionNormalize(rotate)) *
43 XMMatrixTranslation(position.x, position.y, position.z);
44 Matrix4x4 parentInverse = DirectX::XMMatrixInverse(nullptr, prevParentMatrix_);
45
46 matrixLocal_ = currWorldMat * parentInverse;
47 DecomposeMatrixImpl(&localPosition_, &localRotate_, &localScale_, matrixLocal_);
48
49 GenerateWorldMatrix(&matrixWorld_);
50 // ワールド行列を計算
51 GenerateWorldRotationMatrix(&matrixWorldRot_); // ワールド回転行列更新
52
53 // ワールド行列からワールド座標系の値を逆算して更新
54 DecomposeMatrixImpl(&position, &rotate, &scale, matrixWorld_);
55
56 prevParentMatrix_ = parentMat;
57}
