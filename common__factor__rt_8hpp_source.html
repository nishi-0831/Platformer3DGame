322 //
323 // The Mixed Binary Euclid Algorithm
324 // Sidi Mohamed Sedjelmaci
325 // Electronic Notes in Discrete Mathematics 35 (2009) 169-176
326 //
327 template <class T>
328 BOOST_CXX14_CONSTEXPR T mixed_binary_gcd(T u, T v) BOOST_GCD_NOEXCEPT(T)
329 {
330 if(gcd_traits<T>::less(u, v))
331 constexpr_swap(u, v);
332
333 unsigned shifts = 0;
334
335 if(u == T(0))
336 return v;
337 if(v == T(0))
338 return u;
339
340 shifts = constexpr_min(gcd_traits<T>::make_odd(u), gcd_traits<T>::make_odd(v));
341
342 while(gcd_traits<T>::less(1, v))
343 {
344 u %= v;
345 v -= u;
346 if(u == T(0))
347 return v << shifts;
348 if(v == T(0))
349 return u << shifts;
350 gcd_traits<T>::make_odd(u);
351 gcd_traits<T>::make_odd(v);
352 if(gcd_traits<T>::less(u, v))
353 constexpr_swap(u, v);
354 }
355 return (v == 1 ? v : u) << shifts;
356 }

362 template <typename SteinDomain>
363 BOOST_CXX14_CONSTEXPR SteinDomain Stein_gcd(SteinDomain m, SteinDomain n) BOOST_GCD_NOEXCEPT(SteinDomain)
364 {
365 BOOST_ASSERT(m >= 0);
366 BOOST_ASSERT(n >= 0);
367 if (m == SteinDomain(0))
368 return n;
369 if (n == SteinDomain(0))
370 return m;
371 // m > 0 && n > 0
372 unsigned d_m = gcd_traits<SteinDomain>::make_odd(m);
373 unsigned d_n = gcd_traits<SteinDomain>::make_odd(n);
374 // odd(m) && odd(n)
375 while (m != n)
376 {
377 if (n > m)
378 constexpr_swap(n, m);
379 m -= n;
380 gcd_traits<SteinDomain>::make_odd(m);
381 }
382 // m == n
383 m <<= constexpr_min(d_m, d_n);
384 return m;
385 }

440template <typename Integer>
441inline BOOST_CXX14_CONSTEXPR Integer gcd(Integer const &a, Integer const &b) BOOST_GCD_NOEXCEPT(Integer)
442{
443 if(a == (std::numeric_limits<Integer>::min)())
444 return a == static_cast<Integer>(0) ? gcd_detail::gcd_traits<Integer>::abs(b) : boost::integer::gcd(static_cast<Integer>(a % b), b);
445 else if (b == (std::numeric_limits<Integer>::min)())
446 return b == static_cast<Integer>(0) ? gcd_detail::gcd_traits<Integer>::abs(a) : boost::integer::gcd(a, static_cast<Integer>(b % a));
447 return gcd_detail::optimal_gcd_select(static_cast<Integer>(gcd_detail::gcd_traits<Integer>::abs(a)), static_cast<Integer>(gcd_detail::gcd_traits<Integer>::abs(b)));
448}
