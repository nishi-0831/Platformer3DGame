131template<class A, class T>
132class sp_array_creator {
133 typedef typename A::value_type element;
134
135 enum {
136 offset = sp_array_offset<T, element>::value
137 };
138
139 typedef typename sp_type_with_alignment<sp_array_alignment<T,
140 element>::value>::type type;

142public:
143 template<class U>
144 sp_array_creator(const U& other, std::size_t size) noexcept
145 : other_(other),
146 size_(sp_objects<type>(offset + sizeof(element) * size)) { }
147
148 T* create() {
149 return reinterpret_cast<T*>(other_.allocate(size_));

152 void destroy(T* base) {
153 other_.deallocate(reinterpret_cast<type*>(base), size_);

161template<class T>
162class BOOST_SYMBOL_VISIBLE sp_array_base
163 : public sp_counted_base {
164 typedef typename T::type allocator;
165
166public:

169 template<class A>
170 sp_array_base(const A& other, type* start, std::size_t size)
171 : state_(other, size) {
172 boost::alloc_construct_n(state_.allocator(),
173 boost::first_scalar(start),
174 state_.size() * sp_array_count<type>::value);
175 }
176
177 template<class A, class U>
178 sp_array_base(const A& other, type* start, std::size_t size, const U& list)
179 : state_(other, size) {
180 enum {
181 count = sp_array_count<type>::value
182 };
183 boost::alloc_construct_n(state_.allocator(),
184 boost::first_scalar(start), state_.size() * count,
185 boost::first_scalar(&list), count);
186 }
187
188 T& state() noexcept {
189 return state_;
190 }
191
192 void dispose() noexcept override {
193 boost::alloc_destroy_n(state_.allocator(),
194 boost::first_scalar(sp_array_start<type>(this)),
195 state_.size() * sp_array_count<type>::value);
196 }
197
198 void destroy() noexcept override {
199 sp_array_creator<allocator, sp_array_base> other(state_.allocator(),
200 state_.size());
201 this->~sp_array_base();
202 other.destroy(this);
203 }

205 void* get_deleter(const sp_typeinfo_&) noexcept override {
206 return 0;

216 }

218private:
219 T state_;
220};

254template<class T, class A>
255inline typename std::enable_if<detail::sp_is_unbounded_array<T>::value, shared_ptr<T> >::type
256allocate_shared(const A& allocator, std::size_t count)
257{
258 typedef typename detail::sp_array_element<T>::type element;
259 typedef typename allocator_rebind<A, element>::type other;
260 typedef detail::sp_array_state<other> state;
261 typedef detail::sp_array_base<state> base;
262 detail::sp_array_result<other, base> result(allocator, count);
263 base* node = result.get();
264 element* start = detail::sp_array_start<element>(node);
265 ::new(static_cast<void*>(node)) base(allocator, start, count);
266 result.release();
267 return shared_ptr<T>(detail::sp_internal_constructor_tag(), start,
268 detail::shared_count(static_cast<detail::sp_counted_base*>(node)));
269}
