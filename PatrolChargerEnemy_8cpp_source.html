253 pTransform_->rotate = pInterpolator_->CalculateRot();
254

268void PatrolChargerEnemy::Charge()
269{
270 // 目標地点
271 // -------------------------------------------------------
272 // 注意
273 // x,z軸のみターゲットの座標を使っている。y軸は自身の座標のまま。
274 // ターゲットとのy座標が異なると空中歩行してしまうから。
275 // 斜面を移動させる場合は修正が必要
276 // ------------------------------------------------------
277 // Vector3 distPos = { pTargetTransform_->position };
278 Vector3 distPos = { pTargetTransform_->position.x, pTransform_->position.y, pTargetTransform_->position.z };
279 Vector3 toTarget = distPos - pTransform_->GetWorldPosition();
280 Vector3 toTargetDir = Vector3::Normalize(toTarget);
281
282 // ターゲットの方を向いて、突進!!!
283 pTransform_->position += toTargetDir * chargeSpeed_ * Time::DeltaTimeF();
284 pTransform_->rotate = Quaternion::LookRotation(toTarget, Vector3::Up());
285
286 float detectionDistance = 0.1f;
287 if (toTarget.Size() <= detectionDistance)
288 {
289 // 一定時間待機してから巡回地点に戻る
290 state_.Change(STATE::WAIT);
291 waitTime_ = waitTimeAfterCharge_;
292 nextState_ = STATE::RETURN_TO_PATROL;
293 }

328bool PatrolChargerEnemy::Search()
329{
330 Vector3 toTarget = pTargetTransform_->GetWorldPosition() - pTransform_->GetWorldPosition();
331
332 // 正面方向とターゲットの内積を計算
333 float dotProduct =
334 DirectX::XMVectorGetX(DirectX::XMVector3Dot(pTransform_->Forward(), Vector3::Normalize(toTarget)));
335 // 角度を算出(ラジアン)
336 float angleRadian = std::acosf(std::clamp(dotProduct, -1.0f, 1.0f));
337 // degreeに直す
338 float angleDegree = DirectX::XMConvertToDegrees(angleRadian);
339
340 // 発見したと判定する角度、距離かどうか
341 if (angleDegree <= foundFOV_ && toTarget.Size() <= foundDistance_)
342 {
343 // 発見
344 // Quaternion targetRotation{ Quaternion::LookRotation(Vector3::Normalize(toTarget),pTransform_->Up()) };
345 // state_.Change(STATE::CHARGE);
346 return true;
347 }
348 return false;
