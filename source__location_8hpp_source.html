88

99 r += buffer;

101 unsigned long co = column();
102
103 if( co )
104 {
105 BOOST_ASSERT_SNPRINTF( buffer, ":%lu", co );
106 r += buffer;
107 }
108

110
111 if( *fn != 0 )

113 r += " in function '";
114 r += fn;
115 r += '\'';
116 }
