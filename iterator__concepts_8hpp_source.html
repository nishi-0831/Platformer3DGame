196 template <typename Iterator1, typename Iterator2>
197 void interop_single_pass_constraints(Iterator1 const& i1, Iterator2 const& i2)
198 {
199 bool b;
200 b = i1 == i2;
201 b = i1 != i2;
202
203 b = i2 == i1;
204 b = i2 != i1;
205 boost::ignore_unused_variable_warning(b);
206 }
207
208 template <typename Iterator1, typename Iterator2>
209 void interop_rand_access_constraints(
210 Iterator1 const& i1, Iterator2 const& i2,
211 boost::random_access_traversal_tag, boost::random_access_traversal_tag)
212 {
213 bool b;
214 typename std::iterator_traits<Iterator2>::difference_type n;
215 b = i1 < i2;
216 b = i1 <= i2;
217 b = i1 > i2;
218 b = i1 >= i2;
219 n = i1 - i2;
220
221 b = i2 < i1;
222 b = i2 <= i1;
223 b = i2 > i1;
224 b = i2 >= i1;
225 n = i2 - i1;
226 boost::ignore_unused_variable_warning(b);
227 boost::ignore_unused_variable_warning(n);
228 }

231 void interop_rand_access_constraints(
232 Iterator1 const&, Iterator2 const&,
233 boost::single_pass_traversal_tag, boost::single_pass_traversal_tag)
