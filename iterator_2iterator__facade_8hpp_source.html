108// iterators whose dereference operators reference the same value
109// for all iterators into the same sequence (like many input
110// iterators) need help with their postfix ++: the referenced
111// value must be read and stored away before the increment occurs
112// so that *a++ yields the originally referenced element and not
113// the next one.
114template< typename Iterator >
115class postfix_increment_proxy
116{
117 using value_type = typename iterator_value< Iterator >::type;
118
119public:
120 explicit postfix_increment_proxy(Iterator const& x) :
121 stored_iterator(x),
122 stored_value(*x)
123 {}
124
125 // Returning a mutable reference allows nonsense like
126 // (*r++).mutate(), but it imposes fewer assumptions about the
127 // behavior of the value_type. In particular, recall that
128 // (*r).mutate() is legal if operator* returns by value.
129 // Provides readability of *r++
130 value_type& operator*() const
131 {
132 return stored_value;
133 }
134
135 // Provides X(r++)
136 operator Iterator const&() const
137 {
138 return stored_iterator;
139 }
140
141 // Provides (r++)->foo()
142 value_type* operator->() const
143 {
144 return std::addressof(stored_value);
145 }
146
147private:
148 Iterator stored_iterator;
149 mutable value_type stored_value;
150};

171//
172// In general, we can't determine that such an iterator isn't
173// writable -- we also need to store a copy of the old iterator so
174// that it can be written into.
175template< typename Iterator >
176class writable_postfix_increment_dereference_proxy
177{
178 friend class writable_postfix_increment_proxy< Iterator >;
179
180 using value_type = typename iterator_value< Iterator >::type;
181
182public:
183 explicit writable_postfix_increment_dereference_proxy(Iterator const& x) :
184 stored_iterator(x),
185 stored_value(*x)
186 {}
187
188 // Provides readability of *r++
189 operator value_type&() const
190 {
191 return this->stored_value;
192 }
193
194 template< typename OtherIterator >
195 writable_postfix_increment_dereference_proxy const&
196 operator=(writable_postfix_increment_dereference_proxy< OtherIterator > const& x) const
197 {
198 typedef typename iterator_value< OtherIterator >::type other_value_type;
199 *this->stored_iterator = static_cast< other_value_type& >(x);
200 return *this;
201 }
202
203 // Provides writability of *r++
204 template< typename T >
205 typename std::enable_if<
206 is_not_writable_postfix_increment_dereference_proxy< T >::value,
207 writable_postfix_increment_dereference_proxy const&
208 >::type operator=(T&& x) const
209 {
210 *this->stored_iterator = static_cast< T&& >(x);
211 return *this;
212 }
213
214private:
215 Iterator stored_iterator;
216 mutable value_type stored_value;
217};

259// A metafunction to choose the result type of postfix ++
260//
261// Because the C++98 input iterator requirements say that *r++ has
262// type T (value_type), implementations of some standard
263// algorithms like lexicographical_compare may use constructions
264// like:
265//
266// *r++ < *s++
267//
268// If *r++ returns a proxy (as required if r is writable but not
269// multipass), this sort of expression will fail unless the proxy
270// supports the operator<. Since there are any number of such
271// operations, we're not going to try to support them. Therefore,
272// even if r++ returns a proxy, *r++ will only return a proxy if
273// *r also returns a proxy.
274template< typename Iterator, typename Value, typename Reference, typename CategoryOrTraversal >
275struct postfix_increment_result
276{
277 using type = mp11::mp_eval_if_not<
278 detail::conjunction<
279 // A proxy is only needed for readable iterators
280 std::is_convertible<
281 Reference,
282 // Use add_lvalue_reference to form `reference to Value` due to
283 // some (strict) C++03 compilers (e.g. `gcc -std=c++03`) reject
284 // 'reference-to-reference' in the template which described in CWG
285 // DR106.
286 // http://www.open-std.org/Jtc1/sc22/wg21/docs/cwg_defects.html#106
287 typename std::add_lvalue_reference< Value const >::type
288 >,
289
290 // No multipass iterator can have values that disappear
291 // before positions can be re-visited
292 detail::negation<
293 detail::is_traversal_at_least< CategoryOrTraversal, forward_traversal_tag >
294 >
295 >,
296 Iterator,
297 mp11::mp_if,
298 is_non_proxy_reference< Reference, Value >,
299 postfix_increment_proxy< Iterator >,
300 writable_postfix_increment_proxy< Iterator >
301 >;
302};

304// operator->() needs special support for input iterators to strictly meet the
305// standard's requirements. If *i is not a reference type, we must still
306// produce an lvalue to which a pointer can be formed. We do that by
307// returning a proxy object containing an instance of the reference object.
308template< typename Reference, typename Pointer >
309struct operator_arrow_dispatch // proxy references
310{
311 struct proxy
312 {
313 explicit proxy(Reference const& x) : m_ref(x) {}
314 Reference* operator->() { return std::addressof(m_ref); }
315 // This function is needed for MWCW and BCC, which won't call
316 // operator-> again automatically per 13.3.1.2 para 8
317 operator Reference*() { return std::addressof(m_ref); }
318 Reference m_ref;
319 };
320
321 using result_type = proxy;
322
323 static result_type apply(Reference const& x)
324 {
325 return result_type(x);
326 }
327};
328
329template< typename T, typename Pointer >
330struct operator_arrow_dispatch< T&, Pointer > // "real" references
331{
332 using result_type = Pointer;
333
334 static result_type apply(T& x)
335 {
336 return std::addressof(x);
337 }

340// A proxy return type for operator[], needed to deal with
341// iterators that may invalidate referents upon destruction.
342// Consider the temporary iterator in *(a + n)
343template< typename Iterator >
344class operator_brackets_proxy
345{
346 // Iterator is actually an iterator_facade, so we do not have to
347 // go through iterator_traits to access the traits.
348 using reference = typename Iterator::reference;
349 using value_type = typename Iterator::value_type;
350
351public:
352 operator_brackets_proxy(Iterator const& iter) :
353 m_iter(iter)
354 {}
355
356 operator reference() const
357 {
358 return *m_iter;
359 }
360
361 operator_brackets_proxy& operator=(value_type const& val)
362 {
363 *m_iter = val;
364 return *this;
365 }
366
367private:
368 Iterator m_iter;

585{
586private:

588 Value, CategoryOrTraversal, Reference, Difference
589 >;
590

594 >;

612 {

628 return *static_cast< Derived* >(this);

752//
753// Comparison operator implementation. The library supplied operators
754// enables the user to provide fully interoperable constant/mutable
755// iterator types. I.e. the library provides all operators
756// for all mutable/constant iterator combinations.
757//
758// Note though that this kind of interoperability for constant/mutable
759// iterators is not required by the standard for container iterators.
760// All the standard asks for is a conversion mutable -> constant.
761// Most standard library implementations nowadays provide fully interoperable
762// iterator implementations, but there are still heavily used implementations
763// that do not provide them. (Actually it's even worse, they do not provide
764// them for only a few iterators.)
765//
766// ?? Maybe a BOOST_ITERATOR_NO_FULL_INTEROPERABILITY macro should
767// enable the user to turn off mixed type operators
768//
769// The library takes care to provide only the right operator overloads.
770// I.e.
771//
772// bool operator==(Iterator, Iterator);
773// bool operator==(ConstIterator, Iterator);
774// bool operator==(Iterator, ConstIterator);
775// bool operator==(ConstIterator, ConstIterator);
776//
777// ...
778//
779// In order to do so it uses c++ idioms that are not yet widely supported
780// by current compiler releases. The library is designed to degrade gracefully
781// in the face of compiler deficiencies. In general compiler
782// deficiencies result in less strict error checking and more obscure
783// error messages, functionality is not affected.
784//
785// For full operation compiler support for "Substitution Failure Is Not An Error"
786// (aka. enable_if) and boost::is_convertible is required.
787//
788// The following problems occur if support is lacking.
789//
790// Pseudo code
791//
792// ---------------
793// AdaptorA<Iterator1> a1;
794// AdaptorA<Iterator2> a2;
795//
796// // This will result in a no such overload error in full operation
797// // If enable_if or is_convertible is not supported
798// // The instantiation will fail with an error hopefully indicating that
799// // there is no operator== for Iterator1, Iterator2
800// // The same will happen if no enable_if is used to remove
801// // false overloads from the templated conversion constructor
802// // of AdaptorA.
803//
804// a1 == a2;
805// ----------------
806//
807// AdaptorA<Iterator> a;
808// AdaptorB<Iterator> b;
809//
810// // This will result in a no such overload error in full operation
811// // If enable_if is not supported the static assert used
812// // in the operator implementation will fail.
813// // This will accidently work if is_convertible is not supported.
814//
815// a == b;
816// ----------------
817//
