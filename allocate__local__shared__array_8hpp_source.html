73template<class T, class A>
74inline typename std::enable_if<detail::sp_is_unbounded_array<T>::value,
75 local_shared_ptr<T> >::type
76allocate_local_shared(const A& allocator, std::size_t count)
77{
78 typedef typename detail::sp_array_element<T>::type element;
79 typedef typename allocator_rebind<A, element>::type other;
80 typedef detail::lsp_array_state<other> state;
81 typedef detail::sp_array_base<state> base;
82 detail::sp_array_result<other, base> result(allocator, count);
83 base* node = result.get();
84 element* start = detail::sp_array_start<element>(node);
85 ::new(static_cast<void*>(node)) base(allocator, start, count);
86 detail::lsp_array_base& local = node->state().base();
87 local.set(node);
88 result.release();
89 return local_shared_ptr<T>(detail::lsp_internal_constructor_tag(), start,
90 &local);
91}
