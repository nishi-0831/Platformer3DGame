28/* Locks/unlocks two RAII-lockable mutexes taking care that locking is done in
29 * a deadlock-avoiding global order and no double locking happens when the two
30 * mutexes are the same.
31 */
32
33template<typename Mutex>
34class scoped_bilock:private noncopyable
35{
36public:
37 scoped_bilock(Mutex& mutex1,Mutex& mutex2):mutex_eq(&mutex1==&mutex2)
38 {
39 bool mutex_lt=std::less<Mutex*>()(&mutex1,&mutex2);
40
41 ::new (static_cast<void*>(&lock1)) scoped_lock(mutex_lt?mutex1:mutex2);
42 if(!mutex_eq)
43 ::new (static_cast<void*>(&lock2)) scoped_lock(mutex_lt?mutex2:mutex1);
44 }
45
46 ~scoped_bilock()
47 {
48 reinterpret_cast<scoped_lock*>(&lock1)->~scoped_lock();
49 if(!mutex_eq)
50 reinterpret_cast<scoped_lock*>(&lock2)->~scoped_lock();
51 }
52
53private:
54 typedef typename Mutex::scoped_lock scoped_lock;
55 typedef typename aligned_storage<
56 sizeof(scoped_lock),
57 alignment_of<scoped_lock>::value
58 >::type scoped_lock_space;
59
60 bool mutex_eq;
61 scoped_lock_space lock1,lock2;
62};
