85mtgb::Vector3 mtgb::RigidBody::GetPushAmount(const DirectX::BoundingSphere& _sphere, const DirectX::BoundingBox& _aabb)
86{
87 Vector3 aabbMin = _aabb.Center - _aabb.Extents;
88 Vector3 aabbMax = _aabb.Center + _aabb.Extents;
89
90 // 最短地点
91 Vector3 closest;
92
93 // 各座標軸にクランプする
94 closest.x = std::clamp(_sphere.Center.x, aabbMin.x, aabbMax.x);
95 closest.y = std::clamp(_sphere.Center.y, aabbMin.y, aabbMax.y);
96 closest.z = std::clamp(_sphere.Center.z, aabbMin.z, aabbMax.z);
97
98 Vector3 v = _sphere.Center - closest;
99 float dist = v.Size();
100
101 // 距離がほぼゼロの場合
102 if (dist <= FLT_EPSILON)
103 {
104 // AABBの中心から球の中心への方向
105 v = Vector3::Normalize(_sphere.Center - _aabb.Center);
106 // 方向が決まらなければ押し出しはしない
107 if (v.Size() <= FLT_EPSILON)
108 return Vector3::Zero();
109 dist = 0.0f;
110 }
111 else
112 {
113 v = Vector3::Normalize(v);
114 }
115
116 // 押し出し量
117 float penetration = _sphere.Radius - dist;
118 if (penetration <= 0.0f)
119 return Vector3::Zero();
120
121 // 最短地点から球の中心へ押し出す
122 return v * penetration;
123}
