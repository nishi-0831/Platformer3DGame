96template< typename Function, typename Input >
97class function_input_iterator :
98 public detail::function_input_iterator_base_t< Function, Input >
99{
100private:
101 using base_type = detail::function_input_iterator_base_t< Function, Input >;
102 using function_arg_type = typename base_type::function_arg_type;
103
104public:
105 using reference = typename base_type::reference;
106
107public:
108 function_input_iterator(function_arg_type f, Input i) :
109 base_type(f, i)
110 {}
111
112 void increment()
113 {
114 if (this->m_value)
115 this->m_value.reset();
116 else
117 (*this->m_f)();
118 ++this->m_state;
119 }
120
121 reference dereference() const
122 {
123 if (!this->m_value)
124 this->m_value = (*this->m_f)();
125 return this->m_value.get();
126 }
127
128 bool equal(function_input_iterator const& other) const
129 {
130 return this->m_f == other.m_f && this->m_state == other.m_state;
131 }
132};
