174 int compare(basic_string_ref x) const {
175 const int cmp = traits::compare ( ptr_, x.ptr_, (std::min)(len_, x.len_));
176 return cmp != 0 ? cmp : ( len_ == x.len_ ? 0 : len_ < x.len_ ? -1 : 1 );

235 size_type find_first_not_of(charT c) const {
236 for ( const_iterator iter = this->cbegin (); iter != this->cend (); ++iter )
237 if ( !traits::eq ( c, *iter ))
238 return std::distance ( this->cbegin (), iter );
239 return npos;
240 }
241
242 size_type find_last_not_of(basic_string_ref s) const {
243 const_reverse_iterator iter = find_not_of ( this->crbegin (), this->crend (), s );
244 return iter == this->crend () ? npos : (this->size() - 1 - std::distance(this->crbegin(), iter));
245 }
246
247 size_type find_last_not_of(charT c) const {
248 for ( const_reverse_iterator iter = this->crbegin (); iter != this->crend (); ++iter )
249 if ( !traits::eq ( c, *iter ))
250 return this->size() - 1 - std::distance(this->crbegin(), iter);
251 return npos;
252 }

256 template <typename Iterator>
257 Iterator find_not_of ( Iterator first, Iterator last, basic_string_ref s ) const {
258 for ( ; first != last ; ++first )
259 if ( 0 == traits::find ( s.ptr_, s.len_, *first ))
260 return first;
261 return last;
262 }
