202void mtgb::ImGuiEditorCamera::DoDolly()
203{
204 Vector3 mouseMove = InputUtil::GetMouseMove();
205 if (mouseMove.Size() != 0)
206 {
207 // カメラの右、上ベクトル
208 Vector3 right = pCameraTransform_->Right();
209 Vector3 up = pCameraTransform_->Up();
210
211 // 移動量を合成
212 Vector3 move = right * -mouseMove.x + up * mouseMove.y;
213
214 pCameraTransform_->position += move * moveSpeed_ * Time::DeltaTimeF();
215 }
216}

218void mtgb::ImGuiEditorCamera::DoPan()
219{
220 Vector3 mouseMove = InputUtil::GetMouseMove();
221 if (mouseMove.Size() != 0)
222 {
223 // マウス移動量を角度に変換
224 azimuthalAngleRad_ -= mouseMove.x * rotateSensitivity_ * Time::DeltaTimeF(); // 水平角度
225
226 polarAngleRad_ += mouseMove.y * rotateSensitivity_ * Time::DeltaTimeF(); // 鉛直角度
227
228 // 鉛直角度を制限
229 polarAngleRad_ = std::clamp(polarAngleRad_, minPolarAngleRad_, maxPolarAngleRad_);
230
231 MoveCameraSphericalOnTheSpot();
232 }

248void mtgb::ImGuiEditorCamera::MoveCameraSphericalOnTheSpot()
249{
250 // ref:https://ja.wikipedia.org/wiki/%E7%90%83%E9%9D%A2%E5%BA%A7%E6%A8%99%E7%B3%BB
251
252 // θ (polar angle) : 鉛直方向
253 float theta = polarAngleRad_;
254
255 // φ (azimuthal angle): 水平方向
256 float phi = azimuthalAngleRad_;
257
258 // 回転中心からのオフセット
259 Vector3 offset;
260
261 // 変換
262 offset.x = sinf(theta) * cos(phi);
263 offset.y = cos(theta);
264 offset.z = sin(theta) * sin(phi);
265
266 // その場回転の時はoffsetの方向を向く
267 pCameraTransform_->rotate = Quaternion::LookRotation(offset, Vector3::Up());
268}

270void mtgb::ImGuiEditorCamera::SelectTransform()
271{
272 Vector3 origin, end, vec;
273 Matrix4x4 proj, view;
274 Game::System<CameraSystem>().GetProjMatrix(&proj);
275 Game::System<CameraSystem>().GetViewMatrix(&view);
276
277 ImGuiWindow* window = ImGui::FindWindowByName(windowName_.c_str());
278
279 if (window == nullptr)
280 return;
281
282 ImRect workRect = window->WorkRect;
283 ImVec2 workPos = workRect.Min;
284 ImGuiUtil::GetMouseRay(
285 origin,
286 end,
287 proj,
288 view,
289 Game::System<ImGuiRenderer>().GetViewport(),
290 { workPos.x, workPos.y }
291 );
292
293 vec = end - origin;
294
295 // vec.Normalize()の結果を別変数に保存して、元の長さを保持
296 Vector3 direction = vec.Normalize(); // これで正規化されたベクトルが返される
297
298 const CameraSystem& camera = Game::System<CameraSystem>();
299 float distance = camera.GetFar() - camera.GetNear(); // 元の長さを計算
300
301 EntityId entityId = Game::System<ColliderCP>().RayCastHitAll(origin, direction, distance);
302 if (entityId != INVALID_ENTITY)
303 {
304 // EntityがTransformコンポーネントを持っていない可能性があるのでTryGet
305 Game::System<TransformCP>().TryGet(pTargetTransform_, entityId);
306
307 mtgb::GameObjectSelectedEvent event { .entityId = entityId };
308 Game::System<EventManager>().GetEvent<mtgb::GameObjectSelectedEvent>().Invoke(event);
309 LOGIMGUI("EditorCamera:Selected");
310 }
311 else
312 {
313 mtgb::GameObjectDeselectedEvent event;
314 if (pTargetTransform_ != nullptr)
315 {
316 event = { .entityId = pTargetTransform_->GetEntityId() };
317 pTargetTransform_ = nullptr;
318 }
319 else
320 {
321 event = { .entityId = INVALID_ENTITY };
322 }
323
324 Game::System<EventManager>().GetEvent<mtgb::GameObjectDeselectedEvent>().Invoke(event);
325 LOGIMGUI("EditorCamera:No Select");
326 }
327}
