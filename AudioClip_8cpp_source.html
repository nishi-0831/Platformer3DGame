68void mtgb::AudioClip::Load(mtbin::MemoryStream& _ms, ComPtr<IXAudio2> _pXAudio2)
69{
70 SAFE_DELETE(pWaveData_);
71 pWaveData_ = new WaveData {};
72 _ms.Seek(0);
73
74 // 先頭4バイトを読み取る
75 byte header[4];
76 _ms.Read(header, sizeof(header), 4);
77
78 // フォーマット判定
79 if (CompareId<4>(header, "RIFF"))
80 {
81 // WAVよみこみ
82
83 _ms.Seek(0);
84 LoadWave(_ms);
85 }
86 else if (CompareId<3>(header, "ID3") || (header[0] == 0xFF && (header[1] == 0xFA || header[1] == 0xFB)))
87 {
88 _ms.Seek(0);
89 LoadMp3(_ms);
90 }
91 else
92 {
93 massert(false && "対応していない音声フォーマットです @AudioClip::Load");
94 }
95
96 _pXAudio2->CreateSourceVoice(&pSourceVoice_, &pWaveData_->waveFormat);
97}

99void mtgb::AudioClip::LoadWave(mtbin::MemoryStream& _ms)
100{
101 // チャンク識別子は 4 byte
102 static const size_t ID_SIZE { 4 };
103
104 // チャンクヘッダ情報
105 struct ChunkHeader
106 {
107 ChunkHeader()
108 : id {}
109 , size { 0 }
110 {
111 }
112
113 byte id[ID_SIZE]; // チャンク識別子
114 uint32_t size; // チャンクボディのサイズ
115 };
116
117 // RIFF チャンク
118 ChunkHeader riff {};
119 _ms.Read(riff.id, sizeof(riff.id), ID_SIZE);
120 massert(CompareId<ID_SIZE>(riff.id, "RIFF") && "RIFF チャンクIdの不一致 @AudioClip::Load");
121 riff.size = _ms.Read<uint32_t>();
122
123 // WAVE チャンク
124 byte wave[ID_SIZE] {};
125 _ms.Read(wave, sizeof(wave), ID_SIZE);
126 massert(CompareId<ID_SIZE>(wave, "WAVE") && "WAVE チャンクIdの不一致 @AudioClip::Load");
127
128 // フォーマットチャンク
129 ChunkHeader format {};
130 _ms.Read(format.id, sizeof(format.id), ID_SIZE);
131 massert(CompareId<ID_SIZE>(format.id, "fmt ") && "フォーマット チャンクIdの不一致 @AudioClip::Load");
132 format.size = _ms.Read<uint32_t>();
133 massert((format.size == 0x10) && "対応していないフォーマットサイズ @AudioClip::Load");
134 _ms.Read(reinterpret_cast<byte*>(&pWaveData_->waveFormat), sizeof(WAVEFORMATEX), format.size);
135
136 // wBitsPerSampleを設定
137 pWaveData_->waveFormat.nBlockAlign = pWaveData_->waveFormat.nChannels * pWaveData_->waveFormat.wBitsPerSample / 8;
138 pWaveData_->waveFormat.nAvgBytesPerSec = pWaveData_->waveFormat.nSamplesPerSec * pWaveData_->waveFormat.nBlockAlign;
139
140 // dataチャンクを探す
141 ChunkHeader header {};
142 while (true)
143 {
144 _ms.Read(header.id, sizeof(header.id), ID_SIZE);
145 header.size = _ms.Read<uint32_t>();
146
147 if (CompareId<ID_SIZE>(header.id, "data"))
148 {
149 break; // dataチャンクに到達したら離脱
150 }
151 else
152 {
153 _ms.Seek(_ms.Current() + header.size); // dataチャンクでないなら読み飛ばし
154 }
155 }
156
157 pWaveData_->bufferSize = header.size;
158 pWaveData_->pBuffer = new byte[header.size] {};
159 _ms.Read(pWaveData_->pBuffer, header.size, header.size);
160}

162void mtgb::AudioClip::LoadMp3(mtbin::MemoryStream& _ms)
163{
164 // MemoryStreamからmp3データを読み込む
165 size_t size = _ms.GetLength();
166 byte* mp3_data = new byte[size];
167 _ms.Read(mp3_data, static_cast<int>(size), static_cast<int>(size));
168
169 // drmp3のデコード初期化
170 drmp3 config;
171 if (!drmp3_init_memory(&config, mp3_data, size, NULL))
172 {
173 massert(false && "MP3の初期化に失敗 @AudioClip::LoadMp3");
174 delete[] mp3_data;
175 }
176
177 // 総PCMサンプル数を取得
178 drmp3_uint64 total_samples = drmp3_get_pcm_frame_count(&config);
179 // バッファサイズを計算
180 size_t pcm_sample_count = (size_t)(total_samples * config.channels);
181 size_t pcm_bytes = pcm_sample_count * sizeof(short);
182
183 // PCM出力バッファ確保
184 pWaveData_->bufferSize = pcm_bytes;
185 pWaveData_->pBuffer = new byte[pcm_bytes];
186
187 // PCMデータを読み込む
188 drmp3_read_pcm_frames_s16(&config, total_samples, reinterpret_cast<short*>(pWaveData_->pBuffer));
189
190 // WAVEフォーマット情報設定
191 WAVEFORMATEX& wf = pWaveData_->waveFormat;
192 wf.wFormatTag = 1;
193 wf.nChannels = static_cast<WORD>(config.channels);
194 wf.nSamplesPerSec = static_cast<DWORD>(config.sampleRate);
195 wf.wBitsPerSample = 16;
196 wf.nBlockAlign = wf.nChannels * wf.wBitsPerSample / 8;
197 wf.nAvgBytesPerSec = wf.nSamplesPerSec * wf.nBlockAlign;
198
199 // 後始末
200 drmp3_uninit(&config);
201 delete[] mp3_data; // メモリリーク防止のために開放
202}
