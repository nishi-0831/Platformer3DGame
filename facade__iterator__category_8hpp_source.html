66//
67// Convert an iterator_facade's traversal category, Value parameter,
68// and ::reference type to an appropriate old-style category.
69//
70// Due to changeset 21683, this now never results in a category convertible
71// to output_iterator_tag.
72//
73// Change at: https://svn.boost.org/trac/boost/changeset/21683
74template< typename Traversal, typename ValueParam, typename Reference >
75struct iterator_facade_default_category
76{
77 using type = typename std::conditional<
78 detail::is_traversal_of_input_iterator< Traversal, ValueParam, Reference >::value,
79 std::input_iterator_tag,
80 Traversal
81 >::type;
82};

84// Specialization for the (typical) case when the reference type is an actual reference
85template< typename Traversal, typename ValueParam, typename Referenced >

87{
88 using type = mp11::mp_cond<

92 detail::is_traversal_of_input_iterator< Traversal, ValueParam, Referenced& >, std::input_iterator_tag,
93 std::true_type, Traversal
94 >;
95};

115//
116// A composite iterator_category tag convertible to Category (a pure
117// old-style category) and Traversal (a pure traversal tag).
118// Traversal must be a strict increase of the traversal power given by
119// Category.
120//
121template< typename Category, typename Traversal >
122struct iterator_category_with_traversal :
123 public Category,
124 public Traversal
125{
126 // Make sure this isn't used to build any categories where
127 // convertibility to Traversal is redundant. Should just use the
128 // Category element in that case.
129 static_assert(
130 !std::is_convertible< iterator_category_to_traversal_t< Category >, Traversal >::value,
131 "Category transformed to corresponding traversal must be convertible to Traversal."
132 );
133
134 static_assert(is_iterator_category< Category >::value, "Category must be an STL iterator category.");
135 static_assert(!is_iterator_category< Traversal >::value, "Traversal must not be an STL iterator category.");
136 static_assert(!is_iterator_traversal< Category >::value, "Category must not be a traversal tag.");
137 static_assert(is_iterator_traversal< Traversal >::value, "Traversal must be a traversal tag.");
138};

140// Computes an iterator_category tag whose traversal is Traversal and
141// which is appropriate for an iterator
142template< typename Traversal, typename ValueParam, typename Reference >
143struct facade_iterator_category_impl
144{
145 static_assert(!is_iterator_category< Traversal >::value, "Traversal must not be an STL iterator category.");
146
147 using category = iterator_facade_default_category_t< Traversal, ValueParam, Reference >;
148
149 using type = typename std::conditional<
150 std::is_same<
151 Traversal,
152 typename iterator_category_to_traversal< category >::type
153 >::value,
154 category,
155 iterator_category_with_traversal< category, Traversal >
156 >::type;
157};
158
159template< typename Traversal, typename ValueParam, typename Reference >
160using facade_iterator_category_impl_t = typename facade_iterator_category_impl< Traversal, ValueParam, Reference >::type;
