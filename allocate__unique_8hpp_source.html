303template<class T, class A>
304class sp_alloc_make {
305public:
306 typedef typename boost::allocator_rebind<A,
307 typename sp_alloc_value<T>::type>::type allocator;
308
309private:
310 typedef boost::alloc_deleter<T, A> deleter;
311
312public:
313 typedef std::unique_ptr<typename sp_alloc_result<T>::type, deleter> type;
314
315 sp_alloc_make(const A& a, std::size_t n)
316 : a_(a)
317 , n_(n)
318 , p_(a_.allocate(n)) { }
319
320 ~sp_alloc_make() {
321 if (p_) {
322 a_.deallocate(p_, n_);
323 }
324 }
325
326 typename allocator::value_type* get() const noexcept {
327 return boost::to_address(p_);
328 }
329
330 allocator& state() noexcept {
331 return a_;
332 }
333
334 type release() noexcept {
335 pointer p = p_;
336 p_ = pointer();
337 return type(typename deleter::pointer(n_, p), deleter(a_));
338 }
339
340private:
341 typedef typename boost::allocator_pointer<allocator>::type pointer;
342
343 allocator a_;
344 std::size_t n_;
345 pointer p_;
346};
