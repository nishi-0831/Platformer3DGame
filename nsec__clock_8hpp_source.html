100template<int Dummy>
101struct QPFHolder
102{
103 static inline double get_nsec_per_tic()
104 {
105 long long freq;
106 //According to MS documentation:
107 //"On systems that run Windows XP or later, the function will always succeed and will thus never return zero"
108 (void)boost::move_detail::QueryPerformanceFrequency(&freq);
109 return double(1000000000.0L / double(freq));
110 }
111
112 static const double nanosecs_per_tic;
113};
114
115template<int Dummy>
116const double QPFHolder<Dummy>::nanosecs_per_tic = get_nsec_per_tic();
117
118inline boost::uint64_t nsec_clock() BOOST_NOEXCEPT
119{
120 double nanosecs_per_tic = QPFHolder<0>::nanosecs_per_tic;
121
122 long long pcount;
123 //According to MS documentation:
124 //"On systems that run Windows XP or later, the function will always succeed and will thus never return zero"
125 (void)boost::move_detail::QueryPerformanceCounter( &pcount );
126 return static_cast<boost::uint64_t>(nanosecs_per_tic * double(pcount));
127}

161inline boost::uint64_t nsec_clock() BOOST_NOEXCEPT
162{
163 struct timespec count;
164 ::clock_gettime(BOOST_MOVE_DETAIL_CLOCK_MONOTONIC, &count);
165 boost::uint64_t r = static_cast<boost::uint64_t>(count.tv_sec);
166 r *= 1000000000U;
167 r += static_cast<boost::uint64_t>(count.tv_nsec);
168 return r;
169}

220inline void cpu_timer::start() BOOST_NOEXCEPT
221{
222 m_is_stopped = false;
223 get_cpu_times(m_times);
224}
225
226inline void cpu_timer::stop() BOOST_NOEXCEPT
227{
228 if (is_stopped())
229 return;
230 m_is_stopped = true;
231
232 cpu_times current;
233 get_cpu_times(current);
234 m_times.wall = (current.wall - m_times.wall);
235 m_times.user = (current.user - m_times.user);
236 m_times.system = (current.system - m_times.system);
237}
238
239inline cpu_times cpu_timer::elapsed() const BOOST_NOEXCEPT
240{
241 if (is_stopped())
242 return m_times;
243 cpu_times current;
244 get_cpu_times(current);
245 current.wall -= m_times.wall;
246 current.user -= m_times.user;
247 current.system -= m_times.system;
248 return current;
249}
250
251inline void cpu_timer::resume() BOOST_NOEXCEPT
252{
253 if (is_stopped())
254 {
255 cpu_times current (m_times);
256 start();
257 m_times.wall -= current.wall;
258 m_times.user -= current.user;
259 m_times.system -= current.system;
260 }
261}
