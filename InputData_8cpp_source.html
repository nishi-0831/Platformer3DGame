212float mtgb::InputUtil::GetAxis(Axis _axis, StickType _stickType, WindowContext _context)
213{
214 const InputData& input = GetInput(_context);
215 float value = 0.0f;
216 switch (_axis)
217 {
218 case Axis::X :
219 if (_stickType == StickType::LEFT)
220 {
221 value = static_cast<float>(input.joyStateCurrent_.lX);
222 }
223 else
224 {
225 if (input.controllerType_ == ControllerType::DUAL_SHOCK)
226 {
227 value = static_cast<float>(input.joyStateCurrent_.lZ);
228 }
229 else if (input.controllerType_ == ControllerType::XBOX)
230 {
231 value = static_cast<float>(input.joyStateCurrent_.lRx);
232 }
233 else
234 {
235 value = static_cast<float>(input.joyStateCurrent_.lZ);
236 }
237 }
238 value /= input.config_.xRange;
239 break;
240
241 case Axis::Y :
242 if (_stickType == StickType::LEFT)
243 {
244 value = static_cast<float>(input.joyStateCurrent_.lY);
245 }
246 else
247 {
248 if (input.controllerType_ == ControllerType::DUAL_SHOCK)
249 {
250 value = static_cast<float>(input.joyStateCurrent_.lRz);
251 }
252 else if (input.controllerType_ == ControllerType::XBOX)
253 {
254 value = -static_cast<float>(input.joyStateCurrent_.lRy);
255 }
256 else
257 {
258 value = static_cast<float>(input.joyStateCurrent_.lRz);
259 }
260 }
261 value /= input.config_.yRange;
262 break;
263 default :
264 return 0.0f;
265 }
266 return input.config_.ApplyDeadZone(value);
267}
