125bool mtgb::Collider::IsHit(const Collider& _other) const
126{
127 DirectX::ContainmentType containmentType = DirectX::DISJOINT;
128 // ステージ同士は接触しないものとする
129 if (colliderTag_ == ColliderTag::STAGE && _other.colliderTag_ == ColliderTag::STAGE)
130 {
131 return false;
132 }
133
134 switch (colliderType_)
135 {
136 case ColliderType::TYPE_SPHERE :
137 switch (_other.colliderType_)
138 {
139 case ColliderType::TYPE_SPHERE :
140 containmentType = computeSphere_.Contains(_other.computeSphere_);
141 break;
142 case ColliderType::TYPE_AABB :
143 containmentType = computeSphere_.Contains(_other.computeBox_);
144 break;
145 case ColliderType::TYPE_OBB :
146 containmentType = computeSphere_.Contains(_other.computeOBB_);
147 break;
148 }
149 break;
150 case ColliderType::TYPE_AABB :
151 switch (_other.colliderType_)
152 {
153 case ColliderType::TYPE_SPHERE :
154 containmentType = computeBox_.Contains(_other.computeSphere_);
155 break;
156 case ColliderType::TYPE_AABB :
157 containmentType = computeBox_.Contains(_other.computeBox_);
158 break;
159 case ColliderType::TYPE_OBB :
160 containmentType = computeBox_.Contains(_other.computeOBB_);
161 break;
162 }
163 break;
164 case ColliderType::TYPE_OBB :
165
166 switch (_other.colliderType_)
167 {
168 case ColliderType::TYPE_SPHERE :
169 containmentType = computeOBB_.Contains(_other.computeSphere_);
170 break;
171 case ColliderType::TYPE_AABB :
172 containmentType = computeOBB_.Contains(_other.computeBox_);
173 break;
174 case ColliderType::TYPE_OBB :
175 containmentType = computeOBB_.Contains(_other.computeOBB_);
176 break;
177 }
178 break;
179 }
180
181 if (containmentType == DirectX::INTERSECTS || containmentType == DirectX::CONTAINS)
182 {
183 return true;
184 }
185
186 return false;
187}

206bool mtgb::Collider::IsHit(const Vector3& _center, float _radius) const
207{
208 static Matrix4x4 matrix {};
209
210 if (colliderType_ == ColliderType::TYPE_SPHERE)
211 {
212 pTransform_->GenerateWorldMatrix(&matrix);
213 Vector3 worldPosition { Vector3(computeSphere_.Center) * matrix };
214
215 // 引数で球を作る
216
217 float distance { (_center - worldPosition).Size() };
218 float hitDistance { computeSphere_.Radius + _radius };
219
220 // 距離が双方の球の半径よりも小さければ当たっている
221 return (distance <= hitDistance);
222 }
223
224 return false;
225}

321void mtgb::Collider::Push(const Collider& _other)
322{
323 std::optional<Intersection::IntersectInfo> info = std::nullopt;
324
325 EntityId sphereTypeEntityId = INVALID_ENTITY;
326
327 switch (colliderType_)
328 {
329 case ColliderType::TYPE_SPHERE :
330 switch (_other.colliderType_)
331 {
332 case ColliderType::TYPE_AABB :
333 info = Intersection::Intersect(computeSphere_, _other.computeBox_);
334 break;
335 case ColliderType::TYPE_OBB :
336 info = Intersection::Intersect(computeSphere_, _other.computeOBB_);
337 break;
338 }
339 sphereTypeEntityId = GetEntityId();
340 break;
341 case ColliderType::TYPE_AABB :
342 switch (_other.colliderType_)
343 {
344 case ColliderType::TYPE_SPHERE :
345 info = Intersection::Intersect(_other.computeSphere_, computeBox_);
346 break;
347 }
348 sphereTypeEntityId = _other.GetEntityId();
349 break;
350 case ColliderType::TYPE_OBB :
351 switch (_other.colliderType_)
352 {
353 case ColliderType::TYPE_SPHERE :
354 info = Intersection::Intersect(_other.computeSphere_, computeOBB_);
355 break;
356 }
357 sphereTypeEntityId = _other.GetEntityId();
358 break;
359 }
360
361 Transform& transform = Transform::Get(sphereTypeEntityId);
362
363 if (info.has_value() == false)
364 return;
365
366 if (info.value().closest.y < transform.position.y)
367 {
368 RigidBody& rigidBody = RigidBody::Get(sphereTypeEntityId);
369 rigidBody.OnGround();
370 }
371 transform.position += info.value().push;
372}
