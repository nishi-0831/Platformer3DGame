110// Metafunction to obtain the type of the tuple whose element types
111// are the reference types of an iterator tuple.
112template< typename IteratorTuple >
113struct tuple_of_references;
114
115template< typename IteratorTuple >
116using tuple_of_references_t = typename tuple_of_references< IteratorTuple >::type;
117
118template< template< typename... > class Tuple, typename... Iterators >
119struct tuple_of_references< Tuple< Iterators... > >
120{
121 // Note: non-variadic Boost.Tuple and Boost.Fusion need special handling
122 // to avoid instantiating iterator traits on the trailing "null" types.
123 // If not that, we could simply do
124 // mp11::mp_transform< iterator_reference_t, IteratorTuple >.
125 using type = Tuple<
126 mp11::mp_eval_if<
127 detail::is_trailing_null_type< Iterators >,
128 Iterators,
129 iterator_reference_t, Iterators
130 >...
131 >;
132};
133
134template< typename Front, typename Tail >
135struct tuple_of_references< tuples::cons< Front, Tail > >
136{
137 using type = tuples::cons<
138 iterator_reference_t< Front >,
139 mp11::mp_eval_if<
140 detail::is_trailing_null_type< Tail >,
141 Tail,
142 detail::tuple_of_references_t, Tail
143 >
144 >;
145};

147// Metafunction to obtain the minimal traversal tag in a list
148// of iterators.
149template< typename IteratorList >
150struct minimum_traversal_category_in_iterator_list;
151
152template< typename IteratorList >
153using minimum_traversal_category_in_iterator_list_t = typename minimum_traversal_category_in_iterator_list< IteratorList >::type;
154
155template< template< typename... > class List, typename... Iterators >
156struct minimum_traversal_category_in_iterator_list< List< Iterators... > >
157{
158 // Note: non-variadic Boost.Tuple and Boost.Fusion need special handling
159 // to avoid instantiating iterator traits on the trailing "null" types.
160 // For such types just use random_access_traversal_tag, which will not
161 // affect the result of min_category.
162 using type = min_category_t<
163 mp11::mp_eval_if<
164 detail::is_trailing_null_type< Iterators >,
165 random_access_traversal_tag,
166 pure_iterator_traversal_t, Iterators
167 >...
168 >;
169};

195// Builds and exposes the iterator facade type from which the zip
196// iterator will be derived.
197//
198template< typename IteratorTuple >
199struct zip_iterator_base
200{
201private:
202 // Reference type is the type of the tuple obtained from the
203 // iterators' reference types.
204 using reference = detail::tuple_of_references_t< IteratorTuple >;
205
206 // Value type is the same as reference type.
207 using value_type = reference;
208
209 // Difference type is the first iterator's difference type
210 using difference_type = iterator_difference_t< mp11::mp_front< IteratorTuple > >;
211
212 // Traversal catetgory is the minimum traversal category in the
213 // iterator tuple.
214 using traversal_category = detail::minimum_traversal_category_in_iterator_list_t< IteratorTuple >;
215
216public:
217 // The iterator facade type from which the zip iterator will
218 // be derived.
219 using type = iterator_facade<
220 zip_iterator< IteratorTuple >,
221 value_type,
222 traversal_category,
223 reference,
224 difference_type
225 >;
226};

228template< typename Reference >
229struct converter
230{
231 template< typename Seq >
232 static Reference call(Seq seq)
233 {
234 using tag = typename fusion::traits::tag_of< Reference >::type;
235 return fusion::convert< tag >(seq);
236 }

239template< typename Reference1, typename Reference2 >
240struct converter< std::pair< Reference1, Reference2 > >
241{
242 using reference = std::pair< Reference1, Reference2 >;
243
244 template< typename Seq >
245 static reference call(Seq seq)
246 {
247 return reference(fusion::at_c< 0 >(seq), fusion::at_c< 1 >(seq));
248 }

254//
255// zip_iterator class definition
256//
257template< typename IteratorTuple >
258class zip_iterator :
259 public detail::zip_iterator_base< IteratorTuple >::type
260{
261 // Typedef super_t as our base class.
262 using super_t = typename detail::zip_iterator_base< IteratorTuple >::type;
263
264 // iterator_core_access is the iterator's best friend.
265 friend class iterator_core_access;
266
267public:
268 // Construction
269 // ============
270
271 // Default constructor
272 zip_iterator() = default;
273
274 // Constructor from iterator tuple
275 zip_iterator(IteratorTuple iterator_tuple) :
276 m_iterator_tuple(iterator_tuple)
277 {}
278
279 // Copy constructor
280 template< typename OtherIteratorTuple, typename = enable_if_convertible_t< OtherIteratorTuple, IteratorTuple > >
281 zip_iterator(zip_iterator< OtherIteratorTuple > const& other) :
282 m_iterator_tuple(other.get_iterator_tuple())
283 {}
284
285 // Get method for the iterator tuple.
286 IteratorTuple const& get_iterator_tuple() const { return m_iterator_tuple; }
287
288private:
289 // Implementation of Iterator Operations
290 // =====================================
291
292 // Dereferencing returns a tuple built from the dereferenced
293 // iterators in the iterator tuple.
294 typename super_t::reference dereference() const
295 {
296 using reference = typename super_t::reference;
297 using gen = detail::converter< reference >;
298 return gen::call(fusion::transform(get_iterator_tuple(), detail::dereference_iterator()));
299 }
300
301 // Two zip iterators are equal if all iterators in the iterator
302 // tuple are equal. NOTE: It should be possible to implement this
303 // as
304 //
305 // return get_iterator_tuple() == other.get_iterator_tuple();
306 //
307 // but equality of tuples currently (7/2003) does not compile
308 // under several compilers. No point in bringing in a bunch
309 // of #ifdefs here.
310 //
311 template< typename OtherIteratorTuple >
312 bool equal(zip_iterator< OtherIteratorTuple > const& other) const
313 {
314 return fusion::equal_to(get_iterator_tuple(), other.get_iterator_tuple());
315 }
316
317 // Advancing a zip iterator means to advance all iterators in the
318 // iterator tuple.
319 void advance(typename super_t::difference_type n)
320 {
321 fusion::for_each(m_iterator_tuple, detail::advance_iterator< typename super_t::difference_type >(n));
322 }
323
324 // Incrementing a zip iterator means to increment all iterators in
325 // the iterator tuple.
326 void increment()
327 {
328 fusion::for_each(m_iterator_tuple, detail::increment_iterator());
329 }
330
331 // Decrementing a zip iterator means to decrement all iterators in
332 // the iterator tuple.
333 void decrement()
334 {
335 fusion::for_each(m_iterator_tuple, detail::decrement_iterator());
336 }
337
338 // Distance is calculated using the first iterator in the tuple.
339 template< typename OtherIteratorTuple >
340 typename super_t::difference_type distance_to(zip_iterator< OtherIteratorTuple > const& other) const
341 {
342 return fusion::at_c< 0 >(other.get_iterator_tuple()) - fusion::at_c< 0 >(this->get_iterator_tuple());
343 }
344
345private:
346 // Data Members
347 // ============
348
349 // The iterator tuple.
350 IteratorTuple m_iterator_tuple;
351};
