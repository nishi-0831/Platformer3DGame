89struct counting_iterator_types

135// A distance calculation policy for wrapped iterators
136template< typename Difference, typename Incrementable1, typename Incrementable2 >
137struct iterator_distance
138{
139 static Difference distance(Incrementable1 x, Incrementable2 y)
140 {
141 return y - x;
142 }
143};

145// A distance calculation policy for wrapped numbers
146template< typename Difference, typename Incrementable1, typename Incrementable2 >
147struct number_distance
148{
149 static Difference distance(Incrementable1 x, Incrementable2 y)
150 {
151 return boost::detail::numeric_distance(x, y);

181 counting_iterator(Incrementable x) :
182 super_t(x)
183 {
184 }
185
186private:
187 reference dereference() const
188 {

190 }
191
192 template< typename OtherIncrementable >
193 difference_type
194 distance_to(counting_iterator< OtherIncrementable, CategoryOrTraversal, Difference > const& y) const
195 {
196 using distance_traits = typename std::conditional<
197 detail::is_numeric< Incrementable >::value,
198 detail::number_distance< difference_type, Incrementable, OtherIncrementable >,
199 detail::iterator_distance< difference_type, Incrementable, OtherIncrementable >
200 >::type;
201
202 return distance_traits::distance(this->base(), y.base());
203 }
204};
