312 while (!queue.empty())

315 auto& func = queue.front().second;

329 isSelected = true;

334 queue.pop();

520void mtgb::MTImGui::DrawCone(
521 const Vector3& _position,
522 const Vector3& _direction,
523 float _fovAngleDegree,
524 float _distance,
525 float _thickness,
526 int _segments
527)
528{
529 // 正規化された方向ベクトル
530 Vector3 forward = Vector3::Normalize(_direction);
531
532 // 円錐の底面の中心
533 Vector3 baseCenter = _position + forward * _distance;
534
535 // 円錐の底面の半径
536 // MEMO: tanθ = 対辺 / 底辺 : 対辺 = 底辺 * tanθ
537 float baseRadius = _distance * std::tanf(DirectX::XMConvertToRadians(_fovAngleDegree / 2.0f));
538
539 // 上下方向を決定
540 Vector3 up =
541 std::fabsf(DirectX::XMVectorGetY(forward)) < 0.9f ? Vector3{0.0f, 1.0f, 0.0f} : Vector3{1.0f, 0.0f, 0.0f};
542
543 // 右方向を計算
544 Vector3 right = Vector3::Normalize(Vector3::Cross(forward, up));
545
546 // 上方向を再計算
547 up = Vector3::Cross(right, forward);
548
549 // 底面の円周上の点を計算
550 std::vector<Vector3> baseCirclePoints(_segments);
551 for (int i = 0; i < _segments; i++)
552 {
553 float angle = DirectX::XM_2PI * i / _segments;
554 float x = baseRadius * std::cosf(angle);
555 float y = baseRadius * std::sinf(angle);
556 baseCirclePoints[i] = baseCenter + right * x + up * y;
557 }
558
559 // 円錐の側面を描画
560 for (int i = 0; i < _segments; i++)
561 {
562 DrawLine(_position, baseCirclePoints[i], _thickness);
563 }
564
565 // 底面の円を描画
566 for (int i = 0; i < _segments; i++)
567 {
568 int nextIdx = (i + 1) % _segments;
569 DrawLine(baseCirclePoints[i], baseCirclePoints[nextIdx], _thickness);
570 }
571}
